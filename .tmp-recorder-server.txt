function errorJson(error, fallback) {
  const payload = { error: error.code || fallback };
  const detail = String(error.gitStderr || error.gitStdout || "").trim();
  if (detail && detail !== payload.error) payload.detail = detail.slice(0, 1200);
  return payload;
}

  return result;
}
